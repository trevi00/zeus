if n == 36:
    from codex_harness.adapters.experience import parse_lesson
    from codex_harness.adapters.store import MemoryStore
    from codex_harness.application.experience import ExperienceClaims
    from codex_harness.domain.model import ContractError
    data = b'---\noccurrences: 308\nevidence: []\n---\nlesson\n'
    old = parse_lesson('harness', 'lessons/a.md', data, {'kind':'observed','revision':None})
    pinned = parse_lesson('harness', 'lessons/a.md', data, {'kind':'pinned','revision':'a'*40})
    claims = ExperienceClaims(MemoryStore())
    claims.record(old, 'sha256:'+'a'*64)
    try:
        claims.record(pinned, 'sha256:'+'b'*64)
    except ContractError as exc:
        result.update(same_id=old['id']==pinned['id'], valid_basis_transition_rejected=str(exc))
    else:
        raise AssertionError('Expected the observed collision did not reproduce')

elif n == 37:
    from codex_harness.adapters.store import MemoryStore
    from codex_harness.application.workflow import Workflow
    from codex_harness.bootstrap import organization
    from test_workflow import assignment
    store = MemoryStore()
    w = Workflow(store, organization())
    task = w.submit(assignment())
    old = w.claim('worker:implementation', 'old-owner', lease_seconds=120)
    w.cancel(task['id'], 'conductor', 'revoke old execution')
    with store.transaction() as tx:
        fence = tx.get('execution_fences','tasks:'+task['id'])
        tx.put('tasks',task['id'],old)  # Partial restore of only the formerly running row.
    w.heartbeat(old)
    completed = w.complete(old, {'summary':'stale holder wrote past durable fence'})
    result.update(fence_generation=fence['generation'], restored_generation=old['generation'],
                  stale_heartbeat_accepted=True, stale_completion=completed['status'])
    assert fence['generation'] > old['generation'] and completed['status']=='succeeded'

elif n == 38:
    import psycopg
    from psycopg import sql
    from psycopg.conninfo import make_conninfo
    from uuid import uuid4
    from codex_harness.adapters.store import PostgresStore
    from codex_harness.bootstrap import database_url
    import test_idempotency_races as fixtures
    # Delay one owned interpreter after import, modelling a slow scheduled child.
    fixtures.CHILD = fixtures.CHILD.replace("pause = spec['pause']",
        "time.sleep(spec.get('review_start_delay', 0))\npause = spec['pause']")
    schema = 'review_race_'+uuid4().hex
    dsn = database_url()
    with psycopg.connect(dsn) as conn:
        conn.execute(sql.SQL('CREATE SCHEMA {}').format(sql.Identifier(schema)))
    try:
        store = PostgresStore(make_conninfo(dsn,options=f'-c search_path={schema},public'))
        store.migrate()
        specs = [{'message':fixtures.incident('review-slow-child',f'occurrence-{i}'),
                  'review_start_delay':2.5 if i==3 else 0} for i in range(4)]
        with tempfile.TemporaryDirectory(prefix='zeus-review-race-') as tmp:
            rows = fixtures.race(store,Path(tmp),'record_incident',specs,unlocked=True)
        counts = [r['occurrences'] for r in rows]
        result.update(backend='real PostgreSQL isolated schema',child_start_delay_seconds=2.5,
                      unlocked_occurrences=counts,expected_by_submitted_test=[1,1,1,1],
                      exact_negative_control_broken=counts != [1,1,1,1])
        assert result['exact_negative_control_broken']
    finally:
        with psycopg.connect(dsn) as conn:
            conn.execute(sql.SQL('DROP SCHEMA {} CASCADE').format(sql.Identifier(schema)))
